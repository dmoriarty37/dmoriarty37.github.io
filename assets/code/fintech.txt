import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D  
import pandas as pd
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, TensorDataset
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_absolute_error, r2_score
from scipy.interpolate import griddata  
import yfinance as yf

seed = 420
np.random.seed(seed)
torch.manual_seed(seed)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

df = pd.read_csv("/Users/davidmoriarty/Downloads/108105_2023_C_options_data.csv")
df["date"] = pd.to_datetime(df["date"])
vix = yf.download("^VIX", start="2023-01-01", progress=False, multi_level_index=False)[["Close"]]
vix.columns = ["^VIX"]
df = df.merge(vix, left_on="date", right_index=True, how="left")

cleaned_df = df[["S", "K", "T", "Price", "^VIX", "Impl_Vol", "date"]].copy()
cleaned_df = cleaned_df[(cleaned_df["Impl_Vol"] < 0.6) & 
                        (cleaned_df["T"] > 29) & 
                        (cleaned_df["T"] < 681)]
cleaned_df["moneyness"] = cleaned_df["K"] / cleaned_df["S"]
cleaned_df = cleaned_df[cleaned_df["moneyness"] > 0.1]

X = cleaned_df[["moneyness", "T", "S", "^VIX"]].values
y = cleaned_df["Impl_Vol"].values

X_train, X_test, y_train, y_test = train_test_split(X, y, train_size=0.01, random_state=seed)
scaler = StandardScaler()
X_train = torch.from_numpy(scaler.fit_transform(X_train)).float().to(device)
X_test = torch.from_numpy(scaler.transform(X_test)).float().to(device)
y_train = torch.from_numpy(y_train).float().to(device)
y_test = torch.from_numpy(y_test).float().to(device)

train_loader = DataLoader(TensorDataset(X_train, y_train), batch_size=64, shuffle=True)

model = nn.Sequential(
    nn.Linear(4, 256), nn.ReLU(),
    nn.Linear(256, 128), nn.ReLU(),
    nn.Linear(128, 1)
).to(device)

optimizer = torch.optim.Adam(model.parameters(), lr=1e-3, weight_decay=1e-4)
loss_fn = nn.MSELoss()

print("Training...")
for epoch in range(40):
    model.train()
    epoch_loss = 0.0
    for xb, yb in train_loader:
        optimizer.zero_grad()
        loss = loss_fn(model(xb).squeeze(), yb)
        loss.backward()
        optimizer.step()
        epoch_loss += loss.item()
    
    if epoch % 5 == 0:
        print(f"Epoch {epoch:02d} | Loss {epoch_loss / len(train_loader):.8f}")

model.eval()
with torch.no_grad():
    y_pred = model(X_test).squeeze().cpu().numpy()
y_true = y_test.cpu().numpy()

mae = mean_absolute_error(y_true, y_pred)
rmse = np.sqrt(np.mean((y_pred - y_true)**2))
r2 = r2_score(y_true, y_pred)
print(f"\nTest - MAE: {mae:.6f} | RMSE: {rmse:.6f} | R²: {r2:.6f}")

moneyness_test = scaler.inverse_transform(X_test.cpu().numpy())[:, 0]
abs_err = np.abs(y_pred - y_true)

fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(8, 12))

ax1.scatter(moneyness_test, abs_err, alpha=0.5, s=1)
ax1.set(xlabel="Moneyness (K/S)", ylabel="Absolute Error", 
        title=f"Error vs Moneyness (MAE={mae:.4f})")
ax1.grid(True, alpha=0.3)

ax2.scatter(y_true, y_pred, alpha=0.5, s=1)
ax2.plot([y_true.min(), y_true.max()], [y_true.min(), y_true.max()], 'r--', lw=2)
ax2.set(xlabel="Actual IV", ylabel="Predicted IV", title=f"Predictions (R²={r2:.4f})")
ax2.grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

from mpl_toolkits.mplot3d import Axes3D

def visualize_volatility_surface(date_str, cleaned_df, model, scaler, device):
  
    # Filter data for specific date
    date_data = cleaned_df[cleaned_df["date"] == date_str].copy()
    
    if len(date_data) == 0:
        print(f"⚠ No data found for {date_str}")
        return None
    
    # Get spot price and VIX for this date
    S_date = date_data["S"].iloc[0]
    vix_date = date_data["^VIX"].iloc[0]
    
    print(f"\n Generating surface for {date_str}")
    print(f"   Spot: ${S_date:.2f} | VIX: {vix_date:.2f}")
    
    # Create grid for smooth surface
    moneyness_range = np.linspace(0.85, 1.20, 100)
    T_range = np.linspace(30, 450, 100)
    moneyness_grid, T_grid = np.meshgrid(moneyness_range, T_range)
    
    # PREDICTED SURFACE
    # Prepare inputs
    n_points = moneyness_grid.size
    X_surface = np.column_stack([
        moneyness_grid.flatten(),
        T_grid.flatten(),
        np.full(n_points, S_date),
        np.full(n_points, vix_date)
    ])
    
    # Predict
    X_scaled = torch.from_numpy(scaler.transform(X_surface)).float().to(device)
    model.eval()
    with torch.no_grad():
        IV_pred = model(X_scaled).squeeze().cpu().numpy()
    IV_grid_pred = IV_pred.reshape(moneyness_grid.shape)
    
    # OBSERVED SURFACE
    # Interpolate observed data onto grid
    obs_moneyness = date_data["moneyness"].values
    obs_T = date_data["T"].values
    obs_IV = date_data["Impl_Vol"].values
    
    points = np.column_stack([obs_moneyness, obs_T])
    IV_grid_obs = griddata(points, obs_IV, (moneyness_grid, T_grid), method='cubic')
    
    #CREATE FIGURE
    fig = plt.figure(figsize=(16, 7))
    
    # Observed surface (left)
    ax1 = fig.add_subplot(121, projection='3d')
    surf1 = ax1.plot_surface(moneyness_grid, T_grid, IV_grid_obs,
                             cmap='turbo', alpha=0.95, edgecolor='none',
                             vmin=0.10, vmax=0.23, antialiased=True)
    ax1.set_xlabel('K/S', fontsize=11, labelpad=8)
    ax1.set_ylabel('T (days)', fontsize=11, labelpad=8)
    ax1.set_zlabel('IV', fontsize=11, labelpad=8)
    ax1.set_title(f'OBS {date_str}', fontsize=14, fontweight='bold', pad=15)
    ax1.view_init(elev=20, azim=225)
    cbar1 = fig.colorbar(surf1, ax=ax1, shrink=0.5, aspect=5, pad=0.1)
    cbar1.set_label('IV', fontsize=10)
    
    # Predicted surface (right)
    ax2 = fig.add_subplot(122, projection='3d')
    surf2 = ax2.plot_surface(moneyness_grid, T_grid, IV_grid_pred,
                             cmap='turbo', alpha=0.95, edgecolor='none',
                             vmin=0.10, vmax=0.23, antialiased=True)
    ax2.set_xlabel('K/S', fontsize=11, labelpad=8)
    ax2.set_ylabel('T (days)', fontsize=11, labelpad=8)
    ax2.set_zlabel('IV', fontsize=11, labelpad=8)
    ax2.set_title(f'PRED {date_str}', fontsize=14, fontweight='bold', pad=15)
    ax2.view_init(elev=20, azim=225)
    cbar2 = fig.colorbar(surf2, ax=ax2, shrink=0.5, aspect=5, pad=0.1)
    cbar2.set_label('IV', fontsize=10)
    
    fig.suptitle('Visualizing the Volatility Surface', fontsize=16, fontweight='bold', y=0.98)
    plt.tight_layout()
    
    return fig
