# ── Keyword matching ──

STYLE_KEYWORDS = {
    "clean": ["clean", "minimal", "simple", "understated", "classic", "sleek", "timeless", "subtle", "basic", "neutral"],
    "retro": ["retro", "vintage", "90s", "80s", "old school", "throwback", "nostalgic", "heritage", "classic"],
    "streetwear": ["street", "streetwear", "urban", "hype", "bold", "edgy", "chunky", "y2k", "skate"],
    "premium": ["premium", "luxury", "quality", "high-end", "designer", "expensive", "refined", "upscale", "quiet luxury", "dad shoe"],
}

OCCASION_KEYWORDS = {
    "casual": ["class", "school", "coffee", "campus", "chill", "relax", "hang", "casual", "friends"],
    "night-out": ["night", "evening", "bar", "club", "date", "dinner", "party", "going out", "drinks"],
    "everyday": ["daily", "all day", "versatile", "everything", "all purpose", "walking", "errand", "everyday", "general"],
}


def detect_keywords(text, keyword_map):
    text_lower = text.lower()
    scores = {}
    for category, words in keyword_map.items():
        score = sum(1 for w in words if w in text_lower)
        if score > 0:
            scores[category] = score
    return scores


def extract_budget(text):
    matches = re.findall(r'\$?(\d+)', text)
    if matches:
        return max(int(m) for m in matches)
    return None


def score_shoe(shoe, style_scores, occasion_scores, budget):
    total = 0
    shoe_tags = " ".join(shoe["style_tags"] + shoe["occasion_tags"]).lower()

    for style, weight in style_scores.items():
        for kw in STYLE_KEYWORDS.get(style, []):
            if kw in shoe_tags:
                total += weight * 2

    for occ, weight in occasion_scores.items():
        for kw in OCCASION_KEYWORDS.get(occ, []):
            if kw in shoe_tags:
                total += weight * 2

    if budget:
        if shoe["price"] <= budget:
            total += 2
        else:
            total -= 5

    return total


def get_outfits(style, occasion):
    occ = occasion if occasion in OUTFITS else "casual"
    sty = style if style in OUTFITS.get(occ, {}) else "clean"
    return OUTFITS.get(occ, {}).get(sty, OUTFITS["casual"]["clean"])


def save_profile(profile):
    profiles = []
    if os.path.exists(PROFILES_PATH):
        with open(PROFILES_PATH) as f:
            try:
                profiles = json.load(f)
            except (json.JSONDecodeError, ValueError):
                profiles = []
    profiles.append(profile)
    with open(PROFILES_PATH, "w") as f:
        json.dump(profiles, f, indent=2)


def load_returning_user():
    if os.path.exists(PROFILES_PATH):
        with open(PROFILES_PATH) as f:
            try:
                profiles = json.load(f)
                if profiles:
                    return profiles[-1]
            except (json.JSONDecodeError, ValueError):
                pass
    return None
