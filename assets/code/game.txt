// Function to set the terminal to non-canonical mode (no Enter key required)
void setTerminalToNonCanonicalMode() {
    struct termios term;
    
    // Get the current terminal attributes
    tcgetattr(STDIN_FILENO, &term);
    
    // Disable canonical mode and set other flags for non-blocking input
    term.c_lflag &= ~ICANON;  // Disable canonical mode
    term.c_lflag &= ~ECHO;    // Disable echo (optional)
    term.c_cc[VMIN] = 1;      // Minimum number of characters to read (1 character)
    term.c_cc[VTIME] = 0;     // Timeout for reading characters (0 means no timeout)
    
    // Apply the new settings
    tcsetattr(STDIN_FILENO, TCSANOW, &term);
}

// Function to restore the terminal to its original settings
void restoreTerminal() {
    struct termios term;
    
    // Restore the terminal settings to default
    tcgetattr(STDIN_FILENO, &term);
    term.c_lflag |= ICANON;   // Re-enable canonical mode
    term.c_lflag |= ECHO;     // Re-enable echo (optional)
    tcsetattr(STDIN_FILENO, TCSANOW, &term);
}

// Function to get non-blocking input with a timeout of 0.2 seconds
char getInputWithTimeout() {
    struct timeval tv;
    fd_set readfds;
    char input = '\0';

    // Set timeout to 0.4 seconds (200 milliseconds)
    tv.tv_sec = 0;
    tv.tv_usec = 200000; // 400,000 microseconds = 1 seconds

    // Set up the file descriptor set for stdin
    FD_ZERO(&readfds);
    FD_SET(STDIN_FILENO, &readfds);

    // Use select() to check for input availability with a timeout
    int result = select(STDIN_FILENO + 1, &readfds, NULL, NULL, &tv);

    if (result > 0) {
        // If input is available, read it
        read(STDIN_FILENO, &input, 1);
    }

    return input;
}
